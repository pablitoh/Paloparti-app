type GroupWithRelations = NonNullable<
  Awaited<ReturnType<PrismaClient['group']['findUnique']>>
> & {
  matches: NonNullable<Awaited<ReturnType<PrismaClient['match']['findMany']>>>;
  members: Array<
    NonNullable<
      Awaited<ReturnType<PrismaClient['groupMember']['findUnique']>>
    > & {
      user: NonNullable<
        Awaited<ReturnType<PrismaClient['user']['findUnique']>>
      >;
    }
  >;
  creator: NonNullable<Awaited<ReturnType<PrismaClient['user']['findUnique']>>>;
  nextMatchRef?: NonNullable<
    Awaited<ReturnType<PrismaClient['match']['findUnique']>>
  >;
  nextMatchId?: string;
  nextMatchDetails?: MatchInterface;
  message?: string; // Mensaje sobre el estado de la membresía
  userStatus?: string; // Estado del usuario en el grupo
};

    id: string;

    id: string;
    isTeamA: boolean;
    scorerId: string;
    scorerName: string | null;
    scorerAvatar: string | null;
    minute?: number;
  }>;
  createdAt?: Date;
  updatedAt?: Date;
  groupId?: string;
  goalsA?: Array<{
    id: string;
    isTeamA: boolean;
    scorerId: string;
    scorerName: string | null;
    scorerAvatar: string | null;
    minute?: number;
  }>;
  goalsB?: Array<{
    id: string;
    isTeamA?: boolean;
    isTeamB?: boolean;
    scorerId: string;
    scorerName: string | null;
    scorerAvatar: string | null;
    minute?: number;
  }>;
}

// Add a custom toast configuration function
const showToast = (message: string, type: 'success' | 'error') => {
  // Save current scroll position
  const scrollPosition = window.scrollY;

  // Configure toast to be less intrusive
  toast[type](message, {
    duration: 3000,
    position: 'bottom-right',
    style: {
      padding: '12px',
      borderRadius: '8px',
      background: type === 'success' ? '#10B981' : '#EF4444',
      color: 'white',
      fontWeight: '500',
      boxShadow: '0 4px 6px rgba(0, 0, 0, 0.1)',
    },
    id: `toast-${Date.now()}`, // Ensure unique ID for each toast
  });

  // After toast appears, restore scroll position
  setTimeout(() => {
    window.scrollTo({
      top: scrollPosition,
      behavior: 'auto',
    });
  }, 100);
};

  // Función para calcular los goleadores y MVPs
  const calculateStats = () => {
    if (!group || !group.matches) return;

    // Calcular goleadores
    const goalsCount: Record<
      string,
      { goals: number; name: string; avatar: string | null }
    > = {};

    // Calcular victorias y partidos jugados para cada jugador
    const playerStats: Record<
      string,
      { wins: number; matches: number; name: string; avatar: string | null }
    > = {};

    // Para cada partido completado
    group.matches.forEach((match: any) => {
      if (match.status !== 'COMPLETED') return;

      // Registrar goles
      if (match.goals && match.goals.length > 0) {
        match.goals.forEach((goal: any) => {
          if (!goalsCount[goal.scorerId]) {
            goalsCount[goal.scorerId] = {
              goals: 0,
              name: goal.scorerName || 'Sin nombre',
              avatar: goal.scorerAvatar,
            };
          }
          goalsCount[goal.scorerId].goals += 1;
        });
      }

      // Determinar el equipo ganador
      const teamAWon = match.scoreA > match.scoreB;
      const isDraw = match.scoreA === match.scoreB;

      // Registrar victorias y partidos para jugadores del Equipo A
      if (match.playersA) {
        match.playersA.forEach((player: any) => {
          if (!playerStats[player.id]) {
            playerStats[player.id] = {
              wins: 0,
              matches: 0,
              name: player.name || 'Sin nombre',
              avatar: player.avatar,
            };
          }
          playerStats[player.id].matches += 1;
          if (teamAWon) playerStats[player.id].wins += 1;
          if (isDraw) playerStats[player.id].wins += 0.5; // Media victoria por empate
        });
      }

      // Registrar victorias y partidos para jugadores del Equipo B
      if (match.playersB) {
        match.playersB.forEach((player: any) => {
          if (!playerStats[player.id]) {
            playerStats[player.id] = {
              wins: 0,
              matches: 0,
              name: player.name || 'Sin nombre',
              avatar: player.avatar,
            };
          }
          playerStats[player.id].matches += 1;
          if (!teamAWon && !isDraw) playerStats[player.id].wins += 1;
          if (isDraw) playerStats[player.id].wins += 0.5; // Media victoria por empate
        });
      }
    });

    // Convertir los objetos a arrays y ordenarlos
    const sortedGoleadores = Object.entries(goalsCount)
      .map(([id, data]) => ({
        id,
        name: data.name,
        goals: data.goals,
        avatar: data.avatar,
      }))
      .sort((a, b) => b.goals - a.goals);

        name: data.name,

        matchesPlayed: data.matches,
        avatar: data.avatar,

      .sort((a, b) => b.winRate - a.winRate);

    setGoleadores(sortedGoleadores);
    setMvps(sortedMvps);

      const tbdPlayersB = tbdPlayers.slice(halfTbdCount);

        'Jugadores reales por equipo - A:',
        realTeamAPlayers.length,
        'B:',
        realTeamBPlayers.length
      );
      console.log(
        'Jugadores TBD por equipo - A:',
        tbdPlayersA.length,
        'B:',
        tbdPlayersB.length
      );

      // Preparar los datos para enviar al endpoint (solo jugadores reales)
      const players = [
        ...realTeamAPlayers.map((p) => ({
          userId: p.id,
          isTeamA: true,
          isPlaceholder: false,
        })),
        ...realTeamBPlayers.map((p) => ({
          userId: p.id,
          isTeamA: false,
          isPlaceholder: false,
        })),
      ];

      // Preparar datos de TBD players para almacenar en la base de datos
      const tbdPlayersData: TbdPlayersData = {
        teamA: tbdPlayersA.map((p) => ({
          id: p.id,
          name: p.name,
          isTeamA: true,
        })),
        teamB: tbdPlayersB.map((p) => ({
          id: p.id,
          name: p.name,
          isTeamA: false,
        })),
      };

      // Llamada a la API para generar equipos
      const response = await fetch(`/api/matches/${group.nextMatchId}/resort`, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
        },
        body: JSON.stringify({
          players: players,
          tbdPlayers: tbdPlayersData,
        }),
      });

      if (!response.ok) {
        const errorData = await response.text();
        console.error('Error response:', errorData);

        let errorMessage = 'Error al sortear equipos';
        try {
          const errorJson = JSON.parse(errorData);
          if (errorJson.message) {
            errorMessage = errorJson.message;
          }
        } catch (e) {
          if (errorData && errorData.length < 200) {
            errorMessage = `Error: ${errorData}`;
          }
        }

        throw new Error(errorMessage);
      }

      const data = await response.json();
      console.log('Respuesta del API resort:', data);

      // Crear una copia del estado actual
      const updatedGroup = { ...group };
      if (!updatedGroup.nextMatchDetails) {
        updatedGroup.nextMatchDetails = {} as MatchInterface;
      }

      // Si tenemos respuesta de la API con datos del partido
      if (data && data.match) {
        const apiMatch = data.match;

        // Mapear los jugadores de la API
        let apiTeamAPlayers: any[] = [];
        let apiTeamBPlayers: any[] = [];

        if (apiMatch.playersA && Array.isArray(apiMatch.playersA)) {
          apiTeamAPlayers = apiMatch.playersA.map((player: any) => {
            return {
              id: player.user?.id || player.userId,
              name: player.user?.name || player.name || 'Jugador',
              avatar: player.user?.image || null,
            };
          });
        }

        if (apiMatch.playersB && Array.isArray(apiMatch.playersB)) {
          apiTeamBPlayers = apiMatch.playersB.map((player: any) => {
            return {
              id: player.user?.id || player.userId,
              name: player.user?.name || player.name || 'Jugador',
              avatar: player.user?.image || null,
            };
          });
        }

        // Combinar jugadores reales con TBD, manteniendo el balance
        const combinedTeamA = [...apiTeamAPlayers, ...tbdPlayersA];
        const combinedTeamB = [...apiTeamBPlayers, ...tbdPlayersB];

        // Actualizar el estado
        updatedGroup.nextMatchDetails.playersA = combinedTeamA;
        updatedGroup.nextMatchDetails.playersB = combinedTeamB;

        // Actualizar otros datos del partido
        if (apiMatch.teamA)
          updatedGroup.nextMatchDetails.teamA = apiMatch.teamA;
        if (apiMatch.teamB)
          updatedGroup.nextMatchDetails.teamB = apiMatch.teamB;

        // Ya no guardamos los TBD players en localStorage, solo en la base de datos
        if (tbdPlayersA.length > 0 || tbdPlayersB.length > 0) {
          localStorage.setItem(
            `tbdPlayers-${group.nextMatchId}`,
            JSON.stringify({
              teamA: tbdPlayersA,
              teamB: tbdPlayersB,
            })
          );
        }
      } else {
        // Fallback si no hay respuesta de la API

          'No se recibieron datos del partido desde la API, usando datos locales'
        );
        updatedGroup.nextMatchDetails.playersA = [
          ...realTeamAPlayers,
          ...tbdPlayersA,
        ];
        updatedGroup.nextMatchDetails.playersB = [
          ...realTeamBPlayers,
          ...tbdPlayersB,
        ];

        localStorage.setItem(
          `tbdPlayers-${group.nextMatchId}`,
          JSON.stringify({
            teamA: tbdPlayersA,
            teamB: tbdPlayersB,
          })
        );
      }

      // Actualizar el estado
      setGroup(updatedGroup);
      showToast('Equipos sorteados exitosamente', 'success');
    } catch (error: any) {
      console.error('Error al sortear equipos:', error);
